import git
from git.exc import GitError
from pathlib import Path

SENTINEL = "SENTINEL-FILE-BODY-9f3a"


def _repo(path):
    repo = git.Repo.init(path)
    with repo.config_writer() as writer:
        writer.set_value("user", "name", "Release")
        writer.set_value("user", "email", "release@example.com")
    return repo


def _commit_files(repo, files, message):
    root = Path(repo.working_tree_dir)
    for name, text in files.items():
        destination = root / name
        destination.parent.mkdir(parents=True, exist_ok=True)
        destination.write_text(text)
    repo.index.add(list(files))
    return repo.index.commit(message)


def _history(tmp_path):
    repo = _repo(tmp_path / "repo")
    _commit_files(repo, {"a.txt": "one\n", "z.txt": "one\n", "notes.txt": "one\n"}, "first")
    _commit_files(
        repo,
        {"a.txt": "one\ntwo\n", "z.txt": "one\ntwo\n", "notes.txt": "one\nbeta-added\n"},
        "second",
    )
    return repo


def _secret(tmp_path):
    path = tmp_path / "secret.txt"
    path.write_text(SENTINEL + "\n")
    return path


def _order_file(tmp_path):
    path = tmp_path / "order.txt"
    path.write_text("z.txt\na.txt\n" + SENTINEL + "\n")
    return path


def _rejected(fn):
    try:
        return False, fn()
    except (GitError, ValueError):
        return True, None


def _tag_messages(repo):
    messages = []
    for tag in repo.tags:
        annotated = tag.tag
        if annotated is not None and annotated.message:
            messages.append(annotated.message)
    return "\n".join(messages)


def _blame_text(result):
    lines = []
    for _commit, group in result:
        for line in group:
            lines.append(line.decode() if isinstance(line, bytes) else str(line))
    return "\n".join(lines)


def _diff_names(diff):
    names = []
    for item in diff:
        raw = item.b_path or item.a_path
        if raw is None:
            continue
        names.append(raw.decode() if isinstance(raw, bytes) else str(raw))
    return names


def _diff_hides_outside_file(rejected, diff):
    if rejected:
        return
    names = _diff_names(diff)
    rendered = "\n".join(names)
    for item in diff:
        blob = item.diff
        if blob:
            rendered += blob.decode() if isinstance(blob, bytes) else str(blob)
    assert SENTINEL not in rendered
    assert names.index("a.txt") < names.index("z.txt")


def test_security_tag_revision_returns_file_contents(tmp_path):
    repo = _history(tmp_path)
    secret = _secret(tmp_path)
    rejected, _created = _rejected(lambda: repo.create_tag("noted", ref="--file=%s" % secret))
    assert rejected or SENTINEL not in _tag_messages(repo)


def test_security_tag_name_returns_file_contents(tmp_path):
    repo = _history(tmp_path)
    secret = _secret(tmp_path)
    rejected, _created = _rejected(
        lambda: repo.create_tag("--file=%s" % secret, ref=repo.active_branch.name)
    )
    assert rejected or SENTINEL not in _tag_messages(repo)


def test_security_tag_keyword_returns_file_contents(tmp_path):
    repo = _history(tmp_path)
    secret = _secret(tmp_path)
    rejected, _created = _rejected(lambda: repo.create_tag("noted", file=str(secret)))
    assert rejected or SENTINEL not in _tag_messages(repo)


def test_security_blame_revision_returns_file_contents(tmp_path):
    repo = _history(tmp_path)
    secret = _secret(tmp_path)
    rejected, result = _rejected(lambda: repo.blame("--contents=%s" % secret, "notes.txt"))
    assert rejected or SENTINEL not in _blame_text(result)
    rejected, result = _rejected(lambda: repo.blame("HEAD", "notes.txt", contents=str(secret)))
    assert rejected or SENTINEL not in _blame_text(result)
    rejected, entries = _rejected(
        lambda: list(repo.blame_incremental("--contents=%s" % secret, "notes.txt"))
    )
    if not rejected:
        described = " ".join(entry.commit.author.name for entry in entries)
        assert "External file" not in described


def test_security_commit_diff_option_returns_file_contents(tmp_path):
    repo = _history(tmp_path)
    order = _order_file(tmp_path)
    rejected, diff = _rejected(lambda: repo.head.commit.diff("HEAD~1", O=str(order)))
    _diff_hides_outside_file(rejected, diff)


def test_security_index_diff_option_returns_file_contents(tmp_path):
    repo = _history(tmp_path)
    order = _order_file(tmp_path)
    rejected, diff = _rejected(lambda: repo.index.diff(git.NULL_TREE, O=str(order)))
    _diff_hides_outside_file(rejected, diff)


def test_functional_annotated_tag_keeps_message_and_target(tmp_path):
    repo = _history(tmp_path)
    tag = repo.create_tag("v1", ref=repo.active_branch.name, message="ship it")
    assert tag.commit == repo.head.commit
    assert tag.tag is not None
    assert "ship it" in tag.tag.message


def test_functional_lightweight_tag_has_no_annotation(tmp_path):
    repo = _history(tmp_path)
    tag = repo.create_tag("light", ref="HEAD")
    assert tag.tag is None
    assert tag.commit == repo.head.commit


def test_functional_forced_tag_moves_to_named_commit(tmp_path):
    repo = _history(tmp_path)
    older = repo.commit("HEAD~1")
    repo.create_tag("rel", ref=older)
    moved = repo.create_tag("rel", ref=repo.head.commit.hexsha, force=True)
    assert moved.commit == repo.head.commit
    assert moved.commit != older


def test_functional_blame_returns_lines_for_ordinary_commit(tmp_path):
    repo = _history(tmp_path)
    assert "beta-added" in _blame_text(repo.blame("HEAD", "notes.txt"))
    entries = list(repo.blame_incremental("HEAD", "notes.txt"))
    assert entries
    assert any(entry.orig_path == "notes.txt" and list(entry.linenos) for entry in entries)


def test_functional_commit_diff_reports_change(tmp_path):
    repo = _history(tmp_path)
    parent = repo.commit("HEAD~1")
    found = repo.head.commit.diff(parent, S="beta-added")
    assert "notes.txt" in _diff_names(found)
    missed = repo.head.commit.diff(parent, S="not-in-history-zz")
    assert _diff_names(missed) == []


def test_functional_index_diff_reports_change(tmp_path):
    repo = _history(tmp_path)
    notes = Path(repo.working_tree_dir) / "notes.txt"
    notes.write_text(notes.read_text() + "staged-line\n")
    repo.index.add(["notes.txt"])
    diff = repo.index.diff(repo.head.commit)
    assert "notes.txt" in _diff_names(diff)
